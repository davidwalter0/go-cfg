// Field holds the parsed struct tag information and processing state for a single
// configuration field. It combines Go reflection data with configuration metadata
// extracted from struct tags.
//
// The Field undergoes several processing stages:
//   1. Tag extraction (default, doc, json, etc.)
//   2. Name transformation (CamelCase -> kebab-case for flags, SNAKE_CASE for env)
//   3. Environment variable lookup
//   4. Flag registration
//
// Example struct field:
//   ServerPort int `json:"port" default:"8080" doc:"HTTP server port"`
//
// Results in Field with:
//   Name: "port"
//   FlagName: "server-port" 
//   KeyName: "SERVER_PORT"
//   Default: "8080"
//   Doc: "HTTP server port"
type Field struct {
    reflect.StructField               // Embedded Go reflection data
    FieldPtr         interface{}     // Pointer to the actual field value
    UseFlags         bool            // Whether to register command-line flags
    StructName       string          // Name of the owning struct
    Name             string          // Field name (from json tag or field name)
    KeyName          string          // Environment variable name (UPPER_SNAKE_CASE)
    Default          string          // Default value from struct tag
    EnvText          string          // Value from environment variable
    Short            string          // Short flag name (single character)
    Doc              string          // Field documentation/help text
    FlagName         string          // Command-line flag name (kebab-case)
    Value            string          // Final resolved value (env > default > zero)
    Omit             bool            // Skip field if json:"-" or json:",omit"
    Required         bool            // Field must have a non-zero value
    Depth            int             // Nesting depth in struct hierarchy
    Ignore           bool            // Skip all processing for this field
    Error            error           // Processing error, if any
    Type             string          // Go type name
    Prefix           string          // Prefix for nested struct processing
}
#+end_src

/*
Terminology:

Decoration: The process of adding prefixes to flag and environment variable names
based on struct names and configuration settings.

Prefixing: Adding a consistent prefix to all generated names, typically from
the root struct name or an explicit prefix parameter.

Nesting: Creating hierarchical names that reflect the struct field hierarchy,
e.g., "Database.Host" becomes "--database-host".

FlatNest: A nesting mode that skips the root struct name, creating cleaner
flag names for single-purpose configuration structs.

Field Processing: The multi-stage process of extracting struct tags, generating
names, looking up environment variables, and registering flags.

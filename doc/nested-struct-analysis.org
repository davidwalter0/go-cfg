func (fc *FlagConfig) registerNestedStruct(structValue reflect.Value, parentName, parentFlagName, parentUsage string) error {
    structType := structValue.Type()
    nc := &types.NamingConventions{}
    
    for i := 0; i < structValue.NumField(); i++ {
        field := structValue.Field(i)
        fieldType := structType.Field(i)
        
        // Skip unexported fields
        if !field.CanSet() {
            continue
        }
        
        // Build nested names
        fieldName := fieldType.Name
        nestedFieldName := parentName + "." + fieldName
        nestedFlagName := parentFlagName + "-" + nc.ToKebabCase(fieldName)
        nestedEnvName := fc.buildNestedEnvName(parentName, fieldName)
        
        // Get tag information with nested context
        usage := fc.getNestedUsage(fieldType, parentUsage, fieldName)
        defaultValue := fieldType.Tag.Get("default")
        required := fieldType.Tag.Get("required") == "true"
        
        // Check for environment variable
        envValue := os.Getenv(nestedEnvName)
        if envValue != "" {
            defaultValue = envValue
        }
        
        // Recursive registration
        if err := fc.registerFieldType(field, nestedFieldName, nestedFlagName, usage, defaultValue, required); err != nil {
            return fmt.Errorf("failed to register nested field %s: %w", nestedFieldName, err)
        }
        
        if required {
            fc.Required[nestedFlagName] = true
        }
    }
    
    return nil
}

*** Flag Naming Strategy
- Dot notation: =--database.host= 
- Hyphen notation: =--database-host= (current implementation)
- Underscore notation: =--database_host=

*** Environment Variable Naming Strategy

- Underscore notation: =DATABASE_HOST= (preferred)
- Double underscore notation: =DATABASE__HOST=

func (fc *FlagConfig) buildNestedEnvName(parentName, fieldName string) string {
    nc := &types.NamingConventions{}
    parentEnv := nc.ToUpperSnakeCase(parentName)
    fieldEnv := nc.ToUpperSnakeCase(fieldName)
    
    // Add configurable prefix support
    if fc.EnvPrefix != "" {
        return fc.EnvPrefix + "_" + parentEnv + "_" + fieldEnv
    }
    return parentEnv + "_" + fieldEnv
}

func (fc *FlagConfig) getNestedUsage(fieldType reflect.StructField, parentUsage, fieldName string) string {
    usage := fieldType.Tag.Get("usage")
    if usage == "" {
        usage = fieldType.Tag.Get("help")
    }
    if usage == "" {
        if parentUsage != "" {
            usage = fmt.Sprintf("%s - %s", parentUsage, fieldName)
        } else {
            usage = fmt.Sprintf("%s configuration", fieldName)
        }
    }
    return usage
}

// NestedConfig holds configuration for nested struct handling
type NestedConfig struct {
    MaxDepth        int    // Maximum nesting depth (default: 10)
    FlagSeparator   string // Flag name separator (default: "-") 
    EnvSeparator    string // Env var separator (default: "_")
    FlattenArrays   bool   // Whether to flatten struct arrays
    IgnoreEmbedded  bool   // Whether to ignore embedded structs
}

// RegisterCustomTypesWithNesting registers types with nested struct support
func (fc *FlagConfig) RegisterCustomTypesWithNesting(structPtr interface{}, config *NestedConfig) error {
    if config == nil {
        config = &NestedConfig{
            MaxDepth:      10,
            FlagSeparator: "-",
            EnvSeparator:  "_",
        }
    }
    
    fc.nestingConfig = config
    return fc.registerStructRecursively(structPtr, "", "", 0)
}

func (fc *FlagConfig) registerStructRecursively(structPtr interface{}, prefix, envPrefix string, depth int) error {
    if depth >= fc.nestingConfig.MaxDepth {
        return fmt.Errorf("maximum nesting depth (%d) exceeded", fc.nestingConfig.MaxDepth)
    }
    
    v := reflect.ValueOf(structPtr)
    if v.Kind() != reflect.Ptr || v.Elem().Kind() != reflect.Struct {
        return fmt.Errorf("expected pointer to struct, got %T", structPtr)
    }
    
    structValue := v.Elem()
    structType := structValue.Type()
    nc := &types.NamingConventions{}
    
    for i := 0; i < structValue.NumField(); i++ {
        field := structValue.Field(i)
        fieldType := structType.Field(i)
        
        if !field.CanSet() {
            continue
        }
        
        fieldName := fieldType.Name
        var flagName, envName string
        
        if prefix != "" {
            flagName = prefix + fc.nestingConfig.FlagSeparator + nc.ToKebabCase(fieldName)
            envName = envPrefix + fc.nestingConfig.EnvSeparator + nc.ToUpperSnakeCase(fieldName)
        } else {
            flagName = nc.ToKebabCase(fieldName)
            envName = nc.ToUpperSnakeCase(fieldName)
        }
        
        // Handle nested structs
        if field.Kind() == reflect.Struct {
            err := fc.registerStructRecursively(field.Addr().Interface(), flagName, envName, depth+1)
            if err != nil {
                return err
            }
            continue
        }
        
        // Handle pointer to struct
        if field.Kind() == reflect.Ptr && field.Type().Elem().Kind() == reflect.Struct {
            if field.IsNil() {
                field.Set(reflect.New(field.Type().Elem()))
            }
            err := fc.registerStructRecursively(field.Interface(), flagName, envName, depth+1)
            if err != nil {
                return err
            }
            continue
        }
        
        // Handle leaf fields
        usage := fc.getNestedUsage(fieldType, "", fieldName)
        defaultValue := fieldType.Tag.Get("default")
        required := fieldType.Tag.Get("required") == "true"
        
        // Check environment variable
        if envValue := os.Getenv(envName); envValue != "" {
            defaultValue = envValue
        }
        
        if err := fc.registerFieldType(field, fieldName, flagName, usage, defaultValue, required); err != nil {
            return fmt.Errorf("failed to register field %s: %w", flagName, err)
        }
        
        if required {
            fc.Required[flagName] = true
        }
    }
    
    return nil
}

func (fc *FlagConfig) registerWithCircularCheck(structPtr interface{}, visited map[reflect.Type]bool, depth int) error {
    structType := reflect.TypeOf(structPtr)
    if structType.Kind() == reflect.Ptr {
        structType = structType.Elem()
    }
    
    if visited[structType] {
        return fmt.Errorf("circular reference detected in struct %s", structType.Name())
    }
    
    visited[structType] = true
    defer delete(visited, structType)
    
    return fc.registerStructRecursively(structPtr, "", "", depth)
}

** Challenge 1: Environment Variable Naming Conflicts

*Problem*: Nested structures can create very long environment variable names that may conflict with system limitations.

*Solution*: Implement configurable naming strategies and length limits:

#+begin_src go :tangle /dev/null :results org :main no :package 'discard :mkdirp yes :tangle-mode o0644
func (fc *FlagConfig) truncateEnvName(envName string, maxLength int) string {
    if len(envName) <= maxLength {
        return envName
    }
    
    // Intelligent truncation preserving meaningful parts
    parts := strings.Split(envName, "_")
    if len(parts) <= 2 {
        return envName[:maxLength]
    }
    
    // Keep first and last parts, abbreviate middle
    first := parts[0]
    last := parts[len(parts)-1]
    available := maxLength - len(first) - len(last) - 2 // 2 underscores
    
    if available <= 0 {
        return envName[:maxLength]
    }
    
    // Abbreviate middle parts
    middle := strings.Join(parts[1:len(parts)-1], "_")
    if len(middle) > available {
        middle = middle[:available]
    }
    
    return fmt.Sprintf("%s_%s_%s", first, middle, last)
}

** Challenge 2: Flag Help Text Organization

*Problem*: Nested flags can create overwhelming help text.

*Solution*: Implement hierarchical help organization:

#+begin_src go :tangle /dev/null :results org :main no :package 'discard :mkdirp yes :tangle-mode o0644
func (fc *FlagConfig) GenerateStructuredUsage() string {
    var buf strings.Builder
    flagGroups := make(map[string][]*pflag.Flag)
    
    fc.FlagSet.VisitAll(func(flag *pflag.Flag) {
        parts := strings.Split(flag.Name, "-")
        if len(parts) > 1 {
            group := parts[0]
            flagGroups[group] = append(flagGroups[group], flag)
        } else {
            flagGroups["general"] = append(flagGroups["general"], flag)
        }
    })
    
    for group, flags := range flagGroups {
        buf.WriteString(fmt.Sprintf("\n%s Configuration:\n", strings.Title(group)))
        for _, flag := range flags {
            buf.WriteString(fmt.Sprintf("  --%s\n", flag.Name))
            buf.WriteString(fmt.Sprintf("      %s\n", flag.Usage))
            if flag.DefValue != "" {
                buf.WriteString(fmt.Sprintf("      (default: %s)\n", flag.DefValue))
            }
        }
    }
    
    return buf.String()
}

*Problem*: Recursive reflection can impact performance.

*Solution*: Implement caching and optimization:

#+begin_src go :tangle /dev/null :results org :main no :package 'discard :mkdirp yes :tangle-mode o0644
type StructCache struct {
    fieldMaps map[reflect.Type][]FieldInfo
    mutex     sync.RWMutex
}

type FieldInfo struct {
    FieldPath []string
    Field     reflect.StructField
    FlagName  string
    EnvName   string
}

func (sc *StructCache) analyzeStruct(structType reflect.Type) []FieldInfo {
    sc.mutex.RLock()
    if fields, exists := sc.fieldMaps[structType]; exists {
        sc.mutex.RUnlock()
        return fields
    }
    sc.mutex.RUnlock()
    
    sc.mutex.Lock()
    defer sc.mutex.Unlock()
    
    // Double-check after acquiring write lock
    if fields, exists := sc.fieldMaps[structType]; exists {
        return fields
    }
    
    fields := sc.analyzeStructRecursively(structType, nil)
    sc.fieldMaps[structType] = fields
    return fields
}

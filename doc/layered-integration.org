* Overview

This document provides a complete implementation guide for the layered integration approach between go-cfg and AutoCfg. This approach maintains both systems as separate packages while composing them to leverage their complementary strengths.

* Architecture

** Integration Flow
#+begin_example
Configuration Sources → AutoCfg Discovery → go-cfg Processing → Final Config

Files (Discovery) ──┐
Environment Vars ───┼─→ AutoCfg ──→ Struct Population ──→ go-cfg ──→ Final Config
Command Flags ──────┘                                       ↑
                                                      Flag Processing
                                                      Help Generation
                                                      Validation
#+end_example

** Precedence Order
1. **AutoCfg Stage**: File discovery and loading (=/etc=, =~/.config=, local, =$AUTOCFG_FILENAME=)
2. **go-cfg Stage**: Environment variables and command-line flags override file values
3. **Final Result**: Clean flag naming with comprehensive configuration sources

// loadConfigFiles simulates AutoCfg file discovery and loading
// In real implementation, this would be: autocfg.Configure(config)
func loadConfigFiles(config interface{}) error {
    // Simulate AutoCfg search order:
    // 1. /etc/{program}/config.json
    // 2. ~/.config/{program}/config.json  
    // 3. .{program}.json (current directory)
    // 4. $AUTOCFG_FILENAME override
    
    configFile := os.Getenv("AUTOCFG_FILENAME")
    if configFile == "" {
        // Default search locations (simplified)
        candidates := []string{
            ".config.json",
            "config.json",
        }
        
        for _, candidate := range candidates {
            if _, err := os.Stat(candidate); err == nil {
                configFile = candidate
                break
            }
        }
    }
    
    if configFile != "" {
        log.Printf("Loading configuration from: %s", configFile)
        // In real implementation: return autocfg.LoadDirect(configFile, config)
        return loadJSONConfig(configFile, config)
    }
    
    return nil // No config file found, continue with defaults
}

    if helpText != "" {
        cfg.HelpText(helpText)

// ConfigureWithDefaults provides a simplified API with sensible defaults

    defaultHelp := generateDefaultHelp()
    return LayeredConfigureWithHelp(config, defaultHelp)

package main

import (
    "encoding/json"
    "fmt"
    "log"
    "time"
    "github.com/davidwalter0/go-cfg"
    "github.com/davidwalter0/go-cfg/pkg/integration"

// DemoConfig shows layered integration with complex nested structure
type DemoConfig struct {
    // Service identification
    ServiceName string `json:"service_name" required:"true"`
    Version     string `json:"version" default:"1.0.0"`
    Environment string `json:"environment" default:"development"`
    
    // API server configuration  
    API struct {
        Host      string        `json:"host" default:"localhost"`
        Port      int           `json:"port" default:"8080"`
        Timeout   time.Duration `json:"timeout" default:"30s"`
        EnableTLS bool          `json:"enable_tls" default:"false"`
    } `json:"api"`
    
    // Database configuration
    Database struct {
        Host     string `json:"host" default:"localhost"`
        Port     int    `json:"port" default:"5432"`
        Name     string `json:"name" required:"true"`
        Username string `json:"username" required:"true"`
        Password string `json:"password" required:"true"`
    } `json:"database"`
    
    // Feature flags
    Features map[string]bool `json:"features" default:"auth:true,metrics:true,debug:false"`
    
    // External services
    Services map[string]string `json:"services"`
    
    // Observability
    Metrics struct {
        Enabled  bool   `json:"enabled" default:"true"`
        Endpoint string `json:"endpoint" default:"/metrics"`
        Port     int    `json:"port" default:"9090"`
    } `json:"metrics"`
}

func main() {
    var config DemoConfig
    
    // Enhanced help text explaining layered configuration
    helpText := `
Layered Integration Demo - go-cfg + AutoCfg

This example demonstrates layered configuration loading:

STAGE 1 - File Discovery (AutoCfg patterns):
  Searches standard locations for configuration files:
  • /etc/layered-demo/config.json
  • ~/.config/layered-demo/config.json  
  • .layered-demo.json (current directory)
  • $AUTOCFG_FILENAME (environment override)
  
STAGE 2 - Flag Processing (go-cfg FlatNest):
  Generates clean, intuitive flags:
  • --service-name (required)
  • --api-host, --api-port, --api-timeout
  • --database-host, --database-name (required)
  • --metrics-enabled, --metrics-port

PRECEDENCE ORDER (lowest to highest):
  Configuration Files → Environment Variables → Command Flags

REQUIRED FIELDS:
  --service-name, --database-name, --database-username, --database-password

EXAMPLES:
  # Use defaults with required fields
  ./demo --service-name=api --database-name=mydb \
         --database-username=user --database-password=secret
  
  # Override with environment variables
  export API_HOST=0.0.0.0
  export DATABASE_HOST=prod-db.internal
  ./demo --service-name=api --database-name=mydb \
         --database-username=user --database-password=secret
  
  # Use configuration file
  echo '{"service_name":"api","database":{"name":"mydb","username":"user","password":"secret"}}' > .layered-demo.json
  ./demo
  
  # Load specific config file
  AUTOCFG_FILENAME=/path/to/config.json ./demo
`
    
    // Configure using layered integration
    if err := integration.LayeredConfigureWithHelp(&config, helpText); err != nil {
        log.Fatalf("Configuration failed: %v", err)
    }
    
    // Display loaded configuration
    fmt.Println("=== Layered Integration Demo ===")
    fmt.Printf("Service: %s v%s (%s)\n", config.ServiceName, config.Version, config.Environment)
    fmt.Printf("API: %s:%d (TLS: %v, Timeout: %v)\n", 
        config.API.Host, config.API.Port, config.API.EnableTLS, config.API.Timeout)
    fmt.Printf("Database: %s@%s:%d/%s\n", 
        config.Database.Username, config.Database.Host, config.Database.Port, config.Database.Name)
    
    // Show features
    fmt.Printf("Features: ")
    for feature, enabled := range config.Features {
        if enabled {
            fmt.Printf("%s ", feature)
        }
    }
    fmt.Println()
    
    // Show metrics config
    if config.Metrics.Enabled {
        fmt.Printf("Metrics: enabled on port %d at %s\n", 
            config.Metrics.Port, config.Metrics.Endpoint)
    }
    
    // Show external services
    if len(config.Services) > 0 {
        fmt.Printf("External services: %v\n", config.Services)
    }
    
    // JSON dump for verification
    fmt.Println("\n=== Configuration JSON ===")
    jsonData, err := json.MarshalIndent(config, "", "  ")
    if err != nil {
        log.Printf("JSON marshal error: %v", err)
    } else {
        fmt.Println(string(jsonData))
    }
    
    // Save current configuration
    fmt.Println("\n=== Configuration Persistence ===")
    stor := cfg.NewStor()
    if err := cfg.CopyOut(config, &stor); err != nil {
        log.Printf("Could not copy configuration: %v", err)
    } else if err := stor.Save("layered-demo-current.yaml"); err != nil {
        log.Printf("Could not save configuration: %v", err)
    } else {
        fmt.Println("✓ Current configuration saved to layered-demo-current.yaml")
    }
}

package integration

import (
    "os"
    "testing"
    "time"
)

type TestConfig struct {
    ServiceName string `json:"service_name" required:"true"`
    API struct {
        Host string `json:"host" default:"localhost"`
        Port int    `json:"port" default:"8080"`
    } `json:"api"`
    Database struct {
        Name string `json:"name" required:"true"`
    } `json:"database"`
}

func TestLayeredConfigure(t *testing.T) {
    // Set required environment variables to avoid flag parsing issues
    os.Setenv("SERVICE_NAME", "test-service")
    os.Setenv("DATABASE_NAME", "test-db")
    defer func() {
        os.Unsetenv("SERVICE_NAME")
        os.Unsetenv("DATABASE_NAME")
    }()
    
    var config TestConfig
    
    // Test basic layered configuration
    err := LayeredConfigure(&config)
    if err != nil {
        t.Fatalf("LayeredConfigure failed: %v", err)
    }
    
    // Verify configuration was loaded
    if config.ServiceName != "test-service" {
        t.Errorf("Expected service_name=test-service, got %s", config.ServiceName)
    }
    
    if config.Database.Name != "test-db" {
        t.Errorf("Expected database.name=test-db, got %s", config.Database.Name)
    }
    
    // Verify defaults were applied
    if config.API.Host != "localhost" {
        t.Errorf("Expected api.host=localhost, got %s", config.API.Host)
    }
    
    if config.API.Port != 8080 {
        t.Errorf("Expected api.port=8080, got %d", config.API.Port)
    }
}

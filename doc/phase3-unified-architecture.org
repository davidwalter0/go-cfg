package config

import (
	"github.com/spf13/pflag"
)

// ConfigManager provides unified configuration management
type ConfigManager interface {
	// Core configuration loading
	Load(config interface{}) error
	
	// Builder pattern configuration
	WithDiscovery(mode DiscoveryMode) ConfigManager
	WithFlagStyle(style FlagStyle) ConfigManager
	WithValidation(enabled bool) ConfigManager
	WithEnvironmentPrefix(prefix string) ConfigManager
	
	// Advanced features
	GenerateSample(format string) ([]byte, error)
	GenerateHelp() string
	SaveCurrent(config interface{}, filename string) error
}

// DiscoveryEngine handles configuration file discovery and loading
type DiscoveryEngine interface {
	FindConfigFiles(opts DiscoveryOptions) ([]string, error)
	LoadConfigFile(filename string, config interface{}) error
	MergeConfigs(files []string, config interface{}) error
}

// FlagProcessor handles command line flag processing
type FlagProcessor interface {
	ProcessStruct(config interface{}, flagSet *pflag.FlagSet) error
	GenerateFlags(config interface{}) (*pflag.FlagSet, error)
	ApplyFlagValues(config interface{}, flagSet *pflag.FlagSet) error
}

// NamingStrategy defines how struct fields map to flag names
type NamingStrategy interface {
	FlagName(fieldPath []string) string
	EnvVarName(fieldPath []string) string
	HelpText(field FieldInfo) string
}

** Configuration Options

#+begin_src go :tangle /home/david/org/go-cfg/doc/phase3-options.go :results org :main no :package 'discard :mkdirp yes :tangle-mode o0644
package config

// DiscoveryMode defines how configuration files are discovered and merged
type DiscoveryMode int

const (
	Union DiscoveryMode = 1 << iota  // Merge all found configs
	First                            // Stop on first found
	Direct                           // Load config files directly
	Indirect                         // Support AutoCfg pointer files
)

// FlagStyle defines how flags are generated and named
type FlagStyle int

const (
	FlatNest FlagStyle = iota  // Clean flat naming: --api-host
	Nested                     // Hierarchical naming: --api.host
	Prefixed                   // Prefixed naming: --config-api-host
)

// ConfigOptions provides comprehensive configuration control
type ConfigOptions struct {
	// Discovery settings
	DiscoveryMode    DiscoveryMode
	SearchPaths      []string
	EnvOverride      string
	ProgramName      string
	
	// Flag processing settings
	FlagStyle        FlagStyle
	EnvironmentPrefix string
	RequiredFields   bool
	
	// Output and generation settings
	HelpTemplate     string
	SampleFormat     string
	ValidateOnLoad   bool
}

// DefaultOptions provides sensible defaults for most use cases
func DefaultOptions() ConfigOptions {
	return ConfigOptions{
		DiscoveryMode:    Union | Direct,
		FlagStyle:        FlatNest,
		EnvOverride:      "AUTOCFG_FILENAME",
		RequiredFields:   true,
		ValidateOnLoad:   true,
		SampleFormat:     "json",
	}
}

#+begin_src go :tangle /home/david/org/go-cfg/doc/phase3-unified-api.go :results org :main no :package 'discard :mkdirp yes :tangle-mode o0644
package main

import (
	"log"
	cfg "github.com/davidwalter0/go-cfg/v3/pkg/config"
)

type ServiceConfig struct {
	ServiceName string `json:"service_name" required:"true" help:"Service identifier"`
	Version     string `json:"version" default:"1.0.0" help:"Service version"`
	
	API struct {
		Host      string `json:"host" default:"localhost" help:"API server host"`
		Port      int    `json:"port" default:"8080" help:"API server port"`
		EnableTLS bool   `json:"enable_tls" default:"false" help:"Enable TLS encryption"`
	} `json:"api" help:"API server configuration"`
	
	Database struct {
		Host     string `json:"host" default:"localhost" help:"Database host"`
		Port     int    `json:"port" default:"5432" help:"Database port"`
		Name     string `json:"name" required:"true" help:"Database name"`
		Username string `json:"username" required:"true" help:"Database username"`
		Password string `json:"password" required:"true" help:"Database password"`
	} `json:"database" help:"Database configuration"`
	
	Features map[string]bool   `json:"features" help:"Feature flags"`
	Services map[string]string `json:"services" help:"External service URLs"`
}

func main() {
	var config ServiceConfig
	
	// Simple API (uses defaults)
	if err := cfg.Load(&config); err != nil {
		log.Fatalf("Configuration failed: %v", err)
	}
	
	// Advanced API with full control
	manager := cfg.NewManager().
		WithDiscovery(cfg.Union | cfg.Direct).
		WithFlagStyle(cfg.FlatNest).
		WithValidation(true).
		WithEnvironmentPrefix("MYAPP")
	
	if err := manager.Load(&config); err != nil {
		log.Fatalf("Advanced configuration failed: %v", err)
	}
	
	// Generate sample configuration
	sample, err := manager.GenerateSample("json")
	if err != nil {
		log.Printf("Sample generation failed: %v", err)
	} else {
		log.Printf("Sample config:\n%s", sample)
	}
	
	// Save current configuration
	if err := manager.SaveCurrent(&config, "current-config.yaml"); err != nil {
		log.Printf("Save failed: %v", err)
	}
	
	log.Printf("Service %s v%s configured successfully", config.ServiceName, config.Version)
}
